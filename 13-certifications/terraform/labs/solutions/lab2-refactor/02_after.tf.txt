# ETAPE 2 : remplacez le contenu de main.tf par ce fichier, puis terraform plan
# Resultat attendu : "0 to add, 0 to change, 0 to destroy" (avec des lignes "has moved to"),
# et le fichier out/legacy.txt existe toujours apres apply.
terraform {
  required_providers {
    local = {
      source  = "hashicorp/local"
      version = "~> 2.5"
    }
  }
}

locals {
  configs = {
    a = "config A"
    b = "config B"
  }
}

resource "local_file" "cfg" {
  for_each = local.configs
  filename = "${path.module}/out/${each.key}.txt"
  content  = each.value
}

# Meme adresse -> nouvelle adresse : aucune destruction
moved {
  from = local_file.a
  to   = local_file.cfg["a"]
}

moved {
  from = local_file.b
  to   = local_file.cfg["b"]
}

# Sortir "legacy" du state SANS supprimer le fichier
removed {
  from = local_file.legacy

  lifecycle {
    destroy = false
  }
}
